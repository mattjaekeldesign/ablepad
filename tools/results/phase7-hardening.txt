# --- DEFENDER EXCLUSIONS ---
# Real-time scanning on audio content is a documented cause of buffer overruns:
# Defender opens a file mid-playback and the audio thread blocks on I/O it did
# not schedule. Average CPU stays low; you still hear a click.
defender:
  realtime_monitoring: ON        # deliberately still on. Only these are excluded.
  exclusion_paths:
    - 'C:\LiveRig'                            # NAM models, IRs, sets, the venv
    - 'C:\ProgramData\Ableton'                # the install itself
    - 'C:\Program Files\Common Files\VST3'    # plugins loaded into the audio thread
  exclusion_processes:
    - 'Ableton Live 12 Suite.exe'
#
# THE TRADEOFF, STATED: a process exclusion means nothing Ableton opens is scanned.
# That is the point (plugins, samples, models) and it is also the risk. NAM models
# and plugin installers pulled off the internet should be scanned BEFORE they land
# in C:\LiveRig, because once there they never will be.
